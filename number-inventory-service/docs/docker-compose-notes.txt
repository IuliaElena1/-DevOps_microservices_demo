# ══════════════════════════════════════════════════════════════
#  PART 1 — WHAT IS YAML?
# ══════════════════════════════════════════════════════════════
#
#   YAML is just a way to write configuration in a text file.
#   It stands for "Yet Another Markup Language".
#   Files end in .yml or .yaml — they are the same thing.
#
#   The only rule you need to remember:
#     - use a colon (:) to separate a name from its value
#     - use indentation (spaces) to show that something belongs inside something else
#     - NEVER use tabs — only spaces
#
#   Example — this is YAML:
#
#     name: phone-sor-api
#     port: 8000
#
#   That means: name is "phone-sor-api" and port is 8000.
#   It is the same as JSON { "name": "phone-sor-api", "port": 8000 }
#   but easier to read.


# ══════════════════════════════════════════════════════════════
#  PART 2 — WHAT IS docker-compose?
# ══════════════════════════════════════════════════════════════
#
#   Until now you started ONE container with docker run.
#   But a real application has multiple pieces running at the same time:
#
#     - your FastAPI app
#     - Kafka (the message broker — explained below)
#     - Zookeeper (Kafka's helper — explained below)
#
#   Each piece is its own container. Without docker-compose you would need to:
#     1. Start Zookeeper manually
#     2. Start Kafka manually
#     3. Start your app manually
#     4. Connect them all manually
#
#   With docker-compose you write ONE file (docker-compose.yml) that
#   describes all the pieces, and then run ONE command:
#
#     docker compose up
#
#   That's it. Docker starts everything in the right order and connects them.
#
#   Think of it like this:
#     docker run        = turning on one device
#     docker compose up = turning on the whole setup with one power switch


# ══════════════════════════════════════════════════════════════
#  PART 3 — WHAT IS KAFKA? WHAT IS ZOOKEEPER?
# ══════════════════════════════════════════════════════════════
#
#   KAFKA is a message broker.
#   It sits between services and lets them communicate WITHOUT calling each other directly.
#
#   Instead of:
#     Service A calls Service B directly → if B is down, A crashes too
#
#   With Kafka:
#     Service A sends a message to Kafka → Kafka holds it → Service B reads it when ready
#     If B is down, the message waits in Kafka until B comes back
#
#   Think of Kafka like a post office:
#     - you drop a letter (message) into the post office (Kafka)
#     - the recipient picks it up when they are ready
#     - you don't need to wait for them
#
#   ZOOKEEPER is Kafka's internal manager.
#   Kafka cannot run without Zookeeper — Zookeeper keeps track of
#   which Kafka servers are alive, who is the leader, etc.
#   You don't interact with Zookeeper directly. It just needs to be running.
#
#   So the startup order must always be:
#     1. Zookeeper starts first
#     2. Kafka starts (needs Zookeeper to already be running)
#     3. Your app starts (needs Kafka to already be running)


# ══════════════════════════════════════════════════════════════
#  PART 4 — HOW CONTAINERS TALK TO EACH OTHER
# ══════════════════════════════════════════════════════════════
#
#   When docker-compose starts, it creates a private network automatically.
#   Every container joins that network.
#   Inside that network, each container has a name — its SERVICE NAME from the file.
#
#   So instead of using IP addresses (which change), containers use names:
#     "I want to reach Kafka" → connect to "kafka:9092"
#     "I want to reach Zookeeper" → connect to "zookeeper:2181"
#
#   Docker acts like a phone book: it translates the name to the right container.
#
#   From YOUR laptop (outside the network), you reach them via ports:
#     Your app  → http://localhost:8000
#     Kafka     → localhost:9092


# ══════════════════════════════════════════════════════════════
#  PART 5 — THE docker-compose.yml FILE, LINE BY LINE
# ══════════════════════════════════════════════════════════════

services:
# "services" is the section where you list every container you want to run.
# Everything indented under it is one container.


# ── CONTAINER 1: zookeeper ────────────────────────────────────

  zookeeper:
#   This is the name of the container. You chose this name.
#   Other containers will use this name to reach it: "zookeeper:2181"

    image: confluentinc/cp-zookeeper:7.5.0
#   "image" = which pre-built Docker image to use.
#   Docker downloads it from Docker Hub (hub.docker.com) if not already on your machine.
#   confluentinc = the company that makes this image (Confluent, the Kafka company)
#   cp-zookeeper = the image name (cp = Confluent Platform)
#   7.5.0 = the version

    environment:
#   "environment" = a list of environment variables passed into the container.
#   An environment variable is a setting the program reads when it starts.
#   Same as -e KEY=VALUE in docker run.

      ZOOKEEPER_CLIENT_PORT: 2181
#     The port Zookeeper listens on for connections from Kafka.
#     2181 is the standard Zookeeper port — always this number.
#     Kafka will connect to "zookeeper:2181" using exactly this port.

      ZOOKEEPER_TICK_TIME: 2000
#     The heartbeat interval in milliseconds (2000ms = 2 seconds).
#     Zookeeper uses this to check if connected clients are still alive.
#     2000 is the standard default — you do not need to change this.


# ── CONTAINER 2: kafka ────────────────────────────────────────

  kafka:
#   The name of the Kafka container.

    image: confluentinc/cp-kafka:7.5.0
#   Same company as Zookeeper, but the Kafka image.
#   Must be the same version as Zookeeper (7.5.0) — they need to match.

    depends_on:
      - zookeeper
#   "depends_on" = do not start this container until Zookeeper is running.
#   Without this, Kafka might start before Zookeeper and crash immediately.

    ports:
      - "9092:9092"
#   Publish port 9092 from the container to your laptop.
#   Format: "laptop-port:container-port"
#   9092 is Kafka's standard port.
#   This lets you connect to Kafka from your laptop (outside Docker's network).

    environment:

      KAFKA_BROKER_ID: 1
#     Every Kafka server has a unique ID number. We have one server, so it is 1.

      KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181
#     Tells Kafka where to find Zookeeper.
#     "zookeeper" = the service name defined above — Docker resolves it to that container.
#     2181 = the port Zookeeper listens on (set above with ZOOKEEPER_CLIENT_PORT).

      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092,PLAINTEXT_HOST://localhost:9092
#     Kafka announces two addresses so both containers AND your laptop can reach it:
#     PLAINTEXT://kafka:9092        → for other containers (they use the service name "kafka")
#     PLAINTEXT_HOST://localhost:9092 → for your laptop (you use localhost)
#     Without both, either containers or your laptop cannot connect.

      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: PLAINTEXT:PLAINTEXT,PLAINTEXT_HOST:PLAINTEXT
#     Maps each listener name to a security protocol.
#     PLAINTEXT means no encryption — fine for local development.
#     In production you would use SSL here.

      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
#     Kafka normally copies data across multiple servers for safety.
#     We have only ONE server, so we set replication to 1 (no copies needed).
#     If you leave this at the default (3) with only 1 server, Kafka fails to start.


# ── CONTAINER 3: api ──────────────────────────────────────────

  api:
#   The name of your FastAPI app container.

    build: .
#   Instead of using a pre-built image, BUILD the image from the Dockerfile
#   in the current folder (the dot means "current folder").
#   This is your code — docker-compose builds it fresh every time you run --build.

    ports:
      - "8000:8000"
#   Publish port 8000 so you can reach the API from your laptop at localhost:8000.

    depends_on:
      - kafka
#   Do not start the app until Kafka is running.
#   The app needs Kafka to be ready before it can connect to it.

    environment:
      KAFKA_BOOTSTRAP_SERVERS: kafka:9092
#     Tells the app where Kafka is.
#     "kafka" = the service name above, Docker resolves it.
#     9092 = Kafka's port.
#     The app reads this variable at startup to know where to connect.


# ══════════════════════════════════════════════════════════════
#  PART 6 — COMMANDS YOU NEED TO KNOW
# ══════════════════════════════════════════════════════════════
#
#   All commands run from the folder that contains docker-compose.yml
#   (number-inventory-service/)
#
#   START everything in the background:
#     docker compose up -d
#     (-d = detached, same as in docker run: runs in background, terminal stays free)
#
#   START and watch all logs live (useful for debugging):
#     docker compose up
#     (no -d: logs stream in the terminal; Ctrl+C stops everything)
#
#   SEE what is running:
#     docker compose ps
#
#   SEE logs for one specific service:
#     docker compose logs -f api
#     (-f = follow, keeps streaming new lines)
#
#   STOP everything (containers stop but are not deleted):
#     docker compose stop
#
#   STOP and DELETE all containers:
#     docker compose down
#
#   STOP, DELETE containers AND delete all saved data:
#     docker compose down -v
#     (-v = volumes: deletes the stored data too — careful with this one)
#
#   REBUILD the app image after you changed the code:
#     docker compose up -d --build
#     (--build forces Docker to rebuild the image from the Dockerfile)


# ══════════════════════════════════════════════════════════════
#  PART 7 — FULL ARCHITECTURE EXPLAINED (your docker-compose.yml)
# ══════════════════════════════════════════════════════════════
#
#   Docker Compose is a tool that lets you define and run applications
#   made of multiple Docker containers with a single command.
#
#   While docker run starts ONE container at a time,
#   docker-compose.yml defines an entire ecosystem and starts everything at once:
#
#     docker compose up -d
#
#   Docker Compose automatically creates a shared private network for all services,
#   so containers can reach each other using the service name directly
#   (e.g. the "api" container finds Kafka at "kafka:9092").
#
#   Your file defines 3 interconnected services:
#
#   [ Zookeeper ] <--(coordination)-- [ Kafka Broker ] <--(messages)-- [ FastAPI ]
#    (port 2181)                        (port 9092)                    (port 8000)


# ── SECTION 1: Zookeeper ──────────────────────────────────────
#
#   Zookeeper is required by classic Kafka versions to manage cluster state.
#   It tracks which Kafka servers are alive and coordinates between them.
#   You never interact with it directly — it just needs to be running.
#
#   services:
#     The keyword that starts the list of containers.
#
#   zookeeper:
#     The name you give this service inside the internal Docker network.
#     Other containers use this name to reach it: "zookeeper:2181"
#
#   image: confluentinc/cp-zookeeper:7.5.0
#     Downloads the ready-made Zookeeper image from Docker Hub.
#     (version 7.5.0 made by Confluent — the company behind Kafka)
#
#   environment:
#     Settings / environment variables sent inside the container.
#
#   ZOOKEEPER_CLIENT_PORT: 2181
#     The internal port where Zookeeper listens for connections from Kafka.
#     2181 is the standard Zookeeper port — always this number.
#
#   ZOOKEEPER_TICK_TIME: 2000
#     The base time unit (in milliseconds) used by Zookeeper for
#     health checks and heartbeats (signals that say "I am still alive").
#     2000ms = 2 seconds. This is the standard default.


# ── SECTION 2: Kafka ──────────────────────────────────────────
#
#   The message broker through which events are sent and received between services.
#
#   kafka:
#     The name of the Kafka service in the internal network.
#
#   image: confluentinc/cp-kafka:7.5.0
#     The official Confluent image for Apache Kafka.
#     Must match the Zookeeper version (7.5.0).
#
#   depends_on: / - zookeeper
#     Tells Docker Compose to start Zookeeper FIRST,
#     because Kafka cannot function without it.
#
#   ports: / - "9092:9092"
#     Exposes port 9092 to the outside.
#     Allows applications on your laptop (localhost) to connect to Kafka
#     from outside the Docker network.
#
#   KAFKA_BROKER_ID: 1
#     The unique identifier of this Kafka node in the cluster.
#     We have one node, so the ID is 1.
#
#   KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181
#     Tells Kafka where to find Zookeeper.
#     Notice it uses the service name "zookeeper" directly — Docker resolves it.
#     2181 is the port set above with ZOOKEEPER_CLIENT_PORT.
#
#   KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092,PLAINTEXT_HOST://localhost:9092
#     Critical network configuration. Tells Kafka how it can be reached:
#
#     PLAINTEXT://kafka:9092
#       For other containers inside the Docker network (like your API).
#       They use the service name "kafka".
#
#     PLAINTEXT_HOST://localhost:9092
#       For programs running directly on your laptop, outside Docker.
#       They use "localhost".
#
#   KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: ...
#     Defines that both connections above use an unencrypted protocol (PLAINTEXT).
#     Fine for local development. In production you would use SSL.
#
#   KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
#     Sets the replication factor to 1.
#     Required when you have a single Kafka server locally —
#     otherwise Kafka looks for other servers to copy data to and fails to start.


# ── SECTION 3: API (your FastAPI app) ─────────────────────────
#
#   api:
#     The name of the service for your application.
#
#   build: .
#     Unlike Kafka and Zookeeper (which use pre-built images),
#     for the API you tell Docker to BUILD the image locally
#     from the Dockerfile in the current folder (. = current folder).
#
#   ports: / - "8000:8000"
#     Maps port 8000 from the container to localhost:8000,
#     so you can access Swagger UI in the browser at http://localhost:8000/docs.
#
#   depends_on: / - kafka
#     Tells Docker to wait for the Kafka service to start before starting your app.
#     The app needs Kafka to be ready before it can connect.
#
#   KAFKA_BOOTSTRAP_SERVERS: kafka:9092
#     Sends an environment variable into your FastAPI code.
#     This tells your application that it should connect to Kafka at "kafka:9092".
#     "kafka" = the service name above, Docker resolves it to the right container.


# ══════════════════════════════════════════════════════════════
#  PART 8 — DEEPER EXPLANATIONS WITH REAL-LIFE ANALOGIES
# ══════════════════════════════════════════════════════════════


# ── 1. WHAT IS A CLUSTER? ─────────────────────────────────────
#
#   A cluster is a group of multiple computers (called servers or nodes)
#   that work together as if they were one single large computer.
#
#   Analogy: If you need to move heavy furniture, one person cannot lift it.
#   But a team of 4 people (a cluster) lifts it easily.
#   If one person gets tired and stops, the other 3 keep holding it
#   without the furniture falling.
#
#   In production, Kafka runs on a cluster (3, 5 or 10 servers at the same time).
#   If one server breaks down, the system does not crash —
#   the other servers take over the work.
#
#   On your laptop: you have ONE server (one container). That is fine for development.


# ── 2. WHERE IS THE IMAGE confluentinc/cp-zookeeper:7.5.0? ───
#
#   The image is physically stored on Docker Hub servers (hub.docker.com) —
#   a public site, a kind of "Google Play Store" for Docker.
#
#   confluentinc  = the company/account that created the image (Confluent)
#   cp-zookeeper  = the name of the software package
#   7.5.0         = the exact version
#
#   When you run docker compose up, your computer automatically downloads
#   the image from Docker Hub to your local hard drive —
#   exactly like downloading an app on your phone.
#   Next time you run it, Docker uses the cached copy (no re-download).


# ── 3. WHAT IS A PORT? INTERNAL vs EXTERNAL PORT ─────────────
#
#   Think of your computer as an apartment building.
#   The building address is "localhost" (or your IP).
#   Ports are the apartment numbers (from 1 to 65535).
#
#     Apartment 8000 = your FastAPI app lives here
#     Apartment 9092 = Kafka lives here
#     Apartment 2181 = Zookeeper lives here
#
#   INTERNAL PORT (inside the container):
#   When Docker runs, it creates a "house with a fence" (an isolated environment).
#   The app inside the container listens on an internal port
#   (e.g. apartment 8000 inside the fence).
#   From outside the fence, you cannot reach it directly.
#
#   EXTERNAL PORT — what "ports: - 9092:9092" does:
#   The mapping creates a door in the fence.
#
#     First  9092 = the port on YOUR physical computer (the Host)
#     Second 9092 = the port inside the Docker container
#
#   Diagram:
#
#    Your Computer (Host / Localhost)
#    ┌─────────────────────────────────────────────────────┐
#    │  App (e.g. Postman / Browser)                       │
#    │    │                                                │
#    │    ▼  sends request to localhost:9092               │
#    │  ┌──────────────────────────────────────────────┐   │
#    │  │  Door in the fence (Port Mapping 9092->9092) │   │
#    │  └───────────────────────┬──────────────────────┘   │
#    │                          │ (enters Docker)          │
#    │                          ▼                          │
#    │               ┌─────────────────────┐               │
#    │               │   Container KAFKA   │               │
#    │               │  (internal port 9092│               │
#    │               └─────────────────────┘               │
#    └─────────────────────────────────────────────────────┘
#
#   Why do you need to specify the port when calling?
#   Because "localhost" tells WHICH computer to reach,
#   but the port tells WHICH APPLICATION on that computer to send the message to.


# ── 4. WHAT IS APACHE KAFKA? ─────────────────────────────────
#
#   Apache Kafka is an ultra-fast postman (a Message Broker or Event Bus).
#
#   Analogy: Imagine a factory.
#   One worker (the SoR Service) packages boxes and puts them on a conveyor belt.
#   The worker does not wait to see who needs the box — they just put it on the belt and leave.
#   At the other end, 3 other workers (Notifications Service, Analytics Service)
#   pick up boxes from the belt when they have time.
#
#   The conveyor belt IS Kafka.
#   It receives messages (events) and keeps them in order,
#   so other services can read them when ready.
#   If a service is temporarily down, messages wait in Kafka until it comes back.


# ── 5. WHAT IS KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181? ──────
#
#   Docker Compose creates a private internal network.
#   In this network, Docker works like a phone book:
#   instead of writing complicated IP addresses (like 172.18.0.2:2181),
#   you write the service name directly: "zookeeper".
#
#   This line tells Kafka:
#   "Hey Kafka, when you want to talk to your manager Zookeeper,
#    look for it in the network under the name 'zookeeper' at port 2181."


# ── 6. KAFKA_ADVERTISED_LISTENERS AND PLAINTEXT ───────────────
#
#   Kafka has a quirk: when an application connects to it,
#   Kafka replies: "Hi! To continue the conversation, call me at address X."
#   These lines define what address X is.
#
#   KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092,PLAINTEXT_HOST://localhost:9092
#
#   PLAINTEXT://kafka:9092
#     If another Docker container (like your FastAPI API) asks Kafka where to send messages,
#     Kafka replies: "I am inside the Docker network, find me at kafka:9092."
#
#   PLAINTEXT_HOST://localhost:9092
#     If a program on your physical laptop (outside Docker) asks Kafka,
#     it replies: "You are outside, find me at localhost:9092."
#
#   Without both entries, either containers or your laptop cannot connect.
#
#   WHAT IS PLAINTEXT? (unencrypted protocol)
#
#   Encrypted protocol (SSL/TLS — HTTPS):
#     When you send a password on the internet, it is encoded (e.g. a8f93j21d).
#     Even if someone intercepts the message, they see only unreadable characters.
#
#   Unencrypted protocol (PLAINTEXT):
#     Data travels as plain text (e.g. "password123").
#     Anyone listening on the network can read it.
#
#   KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: PLAINTEXT:PLAINTEXT tells Kafka:
#   "Do not use complicated SSL security certificates right now.
#    We are on my local computer in development mode —
#    use PLAINTEXT (plain text) to keep things simple and fast."
#
#   In production (on a real server), you would use SSL here.


# ── 7. KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1 ─────────────
#
#   When Kafka runs in a large company (in production), it makes backup copies
#   (replication). If you send a message, Kafka saves it on Server 1,
#   but automatically makes a copy on Server 2 and Server 3 (Replication Factor = 3).
#
#   On your laptop, you have ONE Kafka container (one server).
#
#   If you do NOT set this option, Kafka tries by default to make 3 backup copies.
#   It searches for 2 more servers, finds none, and crashes with an error.
#
#   By setting REPLICATION_FACTOR: 1 you tell Kafka:
#   "Keep only one copy of the data (on yourself).
#    Do not look for other servers — there is only you on this laptop."


# ══════════════════════════════════════════════════════════════
#  PART 9 — COMMON QUESTIONS ANSWERED
# ══════════════════════════════════════════════════════════════


# ── Q1: ARE BACKUP SERVERS JUST SITTING THERE DOING NOTHING? ─
#
#   No! All servers work at the same time. None of them sit idle.
#
#   Think of a team of 3 cashiers at a supermarket:
#     - All 3 serve customers in parallel (they share the workload)
#     - Each cashier has a copy of the price list and stock (replication)
#     - If one gets sick and goes home, the store does not close —
#       the other 2 take over their customers
#
#   Same with Kafka: in a cluster of 3 servers (called Brokers),
#   all 3 servers receive and process data every second.
#   But for safety, when a message arrives on Server 1,
#   Kafka instantly copies it to Server 2 and Server 3 as well.
#
#   If Server 1 explodes or loses power, the data is not lost
#   (it exists on Servers 2 and 3) and the application keeps running
#   without any pause or interruption.


# ── Q2: WHERE DO PORT NUMBERS COME FROM? (9092, 8000, 2181) ──
#
#   Port numbers are conventions (standards) set by the creators of each program.
#   Computers use numbers from 1 to 65535 to tell applications apart.
#   To avoid chaos, the IT community agreed on default numbers:
#
#   Program                        Default Port    Who decided
#   ─────────────────────────────────────────────────────────
#   HTTP (normal websites)         80              World standard
#   HTTPS (secure websites)        443             World standard
#   PostgreSQL (database)          5432            PostgreSQL creators
#   Apache Kafka                   9092            Kafka creators
#   Zookeeper                      2181            Zookeeper creators
#   FastAPI / Uvicorn              8000            Python community convention
#
#   WHAT DOES "9092:9092" MEAN?
#   The format in Docker Compose is:  LEFT : RIGHT  =  HOST_PORT : CONTAINER_PORT
#
#   Right side (9092) = the port Kafka listens on inside the container (set by Kafka creators)
#   Left  side (9092) = the port YOU choose to use on your laptop
#                       you kept it the same (9092) for convenience
#
#   You could write "9999:9092" if port 9092 was already taken on your laptop.
#   That means: on your laptop you call port 9999,
#   and Docker redirects the call to port 9092 inside the container.


# ── Q3: WHAT IS "environment"? ───────────────────────────────
#
#   "environment" is a list of settings (variables) you inject into a container
#   when it starts up.
#
#   Think of a Docker container as a sealed box.
#   When you download the Kafka image from the internet,
#   the code inside is identical for everyone.
#   But Kafka needs to know:
#     - "What is my name?"
#     - "Where is Zookeeper?"
#     - "How many copies should I make of each message?"
#
#   Instead of modifying the code inside the application
#   (which is hard and wrong), you use the environment: section
#   to pass these settings from the outside — like post-it notes stuck on the box.
#
#   Real-life analogy:
#   When you buy a new TV (the container), it comes pre-programmed from the factory.
#   But when you turn it on for the first time, it asks you:
#
#     What country are you in?  → COUNTRY=Romania
#     What language do you want? → LANGUAGE=Romanian
#     What is the Wi-Fi password? → WIFI_PASS=1234
#
#   These are environment variables.
#   They tell the application HOW to behave in the environment it runs in —
#   without rewriting its code.
#
#   In docker-compose.yml, this is exactly what you are doing:
#   you are telling Kafka its Wi-Fi password, its country, its language —
#   all the settings it needs to start correctly on YOUR machine.

#   docker stop phone-sor-api && docker rm phone-sor-api
#
#   This is a cleanup command made of two parts joined by && :
#   && means "run the second command only if the first one succeeded"

# ══════════════════════════════════════════════════════════════
#  PART 13 — NEW LINES IN docker-compose.yml EXPLAINED
# ══════════════════════════════════════════════════════════════
#
#   The file was updated with 3 new lines and 1 changed line.
#   Here is what changed and why.

#   NEW PORT: - "29092:29092"
#
#     A second port was added for Kafka. Why?
#     Kafka now uses TWO separate ports for TWO different audiences:
#
#       9092  → for containers INSIDE the Docker network (your API talking to Kafka)
#       29092 → for programs OUTSIDE Docker (your laptop, curl, Kafka tools)
#
#     Keeping them on different ports avoids a conflict where Kafka gets confused
#     about which address to advertise to which caller.

#   NEW LINE: KAFKA_LISTENERS: PLAINTEXT://0.0.0.0:9092,PLAINTEXT_HOST://0.0.0.0:29092
#
#     Tells Kafka which ports to ACTUALLY OPEN and listen on.
#     0.0.0.0 means "listen on all network interfaces" (accept connections from anywhere).
#
#     PLAINTEXT://0.0.0.0:9092     → opens port 9092 for internal Docker traffic
#     PLAINTEXT_HOST://0.0.0.0:29092 → opens port 29092 for external laptop traffic
#
#     Difference from KAFKA_ADVERTISED_LISTENERS:
#       KAFKA_LISTENERS         = which ports Kafka actually opens (binds to)
#       KAFKA_ADVERTISED_LISTENERS = which address Kafka TELLS clients to use
#       Both are needed. One opens the door, the other tells people where the door is.

#   CHANGED LINE: KAFKA_ADVERTISED_LISTENERS
#
#     Before: PLAINTEXT_HOST://localhost:9092
#     After:  PLAINTEXT_HOST://localhost:29092
#
#     The external address now points to port 29092 (matching the new port mapping).
#     Your laptop connects to Kafka via localhost:29092.

#   NEW LINE: KAFKA_INTER_BROKER_LISTENER_NAME: PLAINTEXT
#
#     In a multi-broker cluster, Kafka brokers talk to each other.
#     This line tells them which listener to use for that internal communication.
#     Value is PLAINTEXT → use the internal Docker network listener (port 9092).
#     Required when you define multiple listeners — without it Kafka does not know
#     which one to use for broker-to-broker communication and refuses to start.
